    timer1.Start();

private void timer1_Tick(object sender, EventArgs e)
{
    DateTime now = DateTime.Now;
    string thu = "";

    switch (now.DayOfWeek)
    {
        case DayOfWeek.Monday:
            thu = "Thứ Hai";
            break;
        case DayOfWeek.Tuesday:
            thu = "Thứ Ba";
            break;
        case DayOfWeek.Wednesday:
            thu = "Thứ Tư";
            break;
        case DayOfWeek.Thursday:
            thu = "Thứ Năm";
            break;
        case DayOfWeek.Friday:
            thu = "Thứ Sáu";
            break;
        case DayOfWeek.Saturday:
            thu = "Thứ Bảy";
            break;
        case DayOfWeek.Sunday:
            thu = "Chủ Nhật";
            break;
    }

    label2.Text = $"{thu}, {now:dd/MM/yyyy - hh:mm:ss tt}";
}

using System;

    public partial class MainForm : Form
    {
        public MainForm()
        {
            InitializeComponent();
        }
        int so1, so2, so3;
        Random rand = new Random();
        private void MainForm_Load(object sender, EventArgs e)
        {
            label1.Text = label2.Text = label3.Text;
		//Khi mới vô form và chưa bấm nút quay số thì không được bấm nút chơi lại và hiển thị kết quả
            button1.Enabled = false;
            button3.Enabled = false;
        }

        private void timer1_Tick(object sender, EventArgs e)
        {
            

            so1 = int.Parse(label1.Text);
            so2 = int.Parse(label2.Text);
            so3 = int.Parse(label3.Text);

            

            button1.Enabled = true;
            button3.Enabled = true;
        }

        private void button1_Click(object sender, EventArgs e)
        {
            timer1.Stop();

            label1.Text = "0";
            label2.Text = "0";
            label3.Text = "0";

            so1 = 0;
            so2 = 0;
            so3 = 0;
            

            button3.Enabled = false;   // Tắt nút kết quả
            button2.Enabled = true;
        }

        private void button2_Click(object sender, EventArgs e)
        {
            timer1.Start();
            so1 = rand.Next(0, 10);
            so2 = rand.Next(0, 10);
            so3 = rand.Next(0, 10);

            label1.Text = so1.ToString();
            label2.Text = so2.ToString();
            label3.Text = so3.ToString();
            button1.Enabled = true;
            button3.Enabled = true;

        }    
      
        private void button3_Click(object sender, EventArgs e)
        {
            
            MessageBox.Show("Kết quả là: " + so1 + " " + so2 + " " + so3, "Tìm kết quả");
        }
